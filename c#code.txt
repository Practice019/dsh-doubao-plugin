public static void Exec(Quicker.Public.IStepContext context)
{
    // 1. 线程检查
    if (System.Threading.Thread.CurrentThread.GetApartmentState() != System.Threading.ApartmentState.STA)
    {
        MessageBox.Show("配置错误：\n请将模块【执行线程】设置为 【后台线程 (STA独立线程) / staLongRun】", "Doubao Relay");
        return;
    }

    // 2. 参数读取
    string inputParam = "";
    try {
        inputParam = context.GetVarValue("input") as string;
    } catch { }
    
    inputParam = inputParam?.Trim().ToLower() ?? "";

    if (inputParam == "shutdown" || inputParam == "stop")
    {
        ShutdownRemoteService();
        return; 
    }

    bool isSilent = inputParam == "silent";

    // 3. 启动
    Application.EnableVisualStyles();
    Application.SetCompatibleTextRenderingDefault(false);
    DoubaoForm form = new DoubaoForm(isSilent);
    Application.Run(form); 
}

private static void ShutdownRemoteService()
{
    try 
    {
        var request = WebRequest.Create("http://127.0.0.1:56666/shutdown");
        request.Timeout = 2000; 
        using(var response = request.GetResponse()) { }
        MessageBox.Show("✅ 已发送停止指令。", "Doubao Relay");
    }
    catch { MessageBox.Show("⚠️ 连接服务失败。", "错误"); }
}

    private async Task InitializeCoreWebView2Async()
    {
        try {
            _env = await CoreWebView2Environment.CreateAsync(null, _userDataFolder);
            _controller = await _env.CreateCoreWebView2ControllerAsync(_webPanel.Handle);
            
            // 重要：即使窗口在屏幕外/隐藏，控制器也必须设为可见，否则 WebView2 可能会挂起加载逻辑
            _controller.IsVisible = true; 
            
            _coreWebView = _controller.CoreWebView2;
            ResizeWebView();

            await _coreWebView.AddScriptToExecuteOnDocumentCreatedAsync(GetNetworkInterceptorScript());
            _coreWebView.WebMessageReceived += OnWebMessageReceived;
            _coreWebView.NavigationCompleted += async (s, e) => {
                if (e.IsSuccess) {
                    await _coreWebView.ExecuteScriptAsync(GetDomControlScript());
                    Log("系统就绪！"); 
                    if (_isDebugMode) ShowToast(); 
                }
            };
            _coreWebView.Navigate(TargetUrl);
        } catch (Exception ex) { Log($"CoreWebView2 初始化失败: {ex.Message}"); }
    }

    private void StartHttpServer()
    {
        try {
            if (_httpListener != null) StopHttpServer(); 
            _httpListener = new HttpListener();
            _httpListener.Prefixes.Add($"http://127.0.0.1:{ServerPort}/");
            _httpListener.Start();
            Task.Run(ListenLoop);
            Log($"HTTP 服务监听中: {ServerPort}");
        } catch (Exception ex) { Log($"HTTP 启动失败: {ex.Message}"); }
    }

    private void StopHttpServer()
    {
        try { if (_httpListener != null) { _httpListener.Stop(); _httpListener.Close(); _httpListener = null; } } catch { }
    }

    private async Task ListenLoop()
    {
        var listener = _httpListener;
        if (listener == null) return;
        while (listener != null && listener.IsListening) {
            try { var context = await listener.GetContextAsync(); HandleRequest(context); } catch { break; }
        }
    }

    private void HandleRequest(HttpListenerContext ctx)
    {
        string path = ctx.Request.Url.AbsolutePath.ToLower();
        string method = ctx.Request.HttpMethod.ToUpper();
        ctx.Response.AppendHeader("Access-Control-Allow-Origin", "*");
        ctx.Response.AppendHeader("Access-Control-Allow-Headers", "Content-Type, Authorization");

        if (method == "OPTIONS") { ctx.Response.StatusCode = 200; ctx.Response.Close(); return; }

        if (path == "/shutdown") {
            ResponseText(ctx, "Closing...");
            this.Invoke(new Action(() => {
                Task.Delay(1000).ContinueWith(_ => this.Invoke(new Action(() => this.Close())));
            }));
            return;
        }

        // 处理 /v1/models 端点
        if (path == "/v1/models" && method == "GET") {
            string modelsJson = @"{
                ""data"": [
                    {
                        ""id"": ""doubao-chat"",
                        ""object"": ""model"",
                        ""created"": 1626777600,
                        ""owned_by"": ""bytedance"",
                        ""supported_endpoint_types"": [
                            ""openai""
                        ]
                    }
                ],
                ""object"": ""list"",
                ""success"": true
            }";
            ResponseText(ctx, modelsJson, 200, "application/json");
            return;
        }

        if ((path == "/v1/chat/completions" || path == "/chat") && method == "POST")
        {
            if (_currentContext != null) { ResponseText(ctx, "{\"error\": \"Busy\"}", 429, "application/json"); return; }
            try {
                using (var reader = new StreamReader(ctx.Request.InputStream, Encoding.UTF8)) {
                    string body = reader.ReadToEnd();
                    Log("收到请求");
                    Log(body);
                    bool isStream = body.Contains("\"stream\": true") || body.Contains("\"stream\":true");
                    // 检查是否需要新建对话
                    bool isNewConversation = body.Contains("\"new_conversation\": true") || body.Contains("\"new_conversation\":true");
                    
                    // 如果没有明确指定新对话，但消息历史只有一条系统消息和一条用户消息，也认为是新对话
                    if (!isNewConversation) {
                        try {
                            // 简单检查消息数量，如果只有一条用户消息，则认为是新对话
                            var userMessageMatches = Regex.Matches(body, "\"role\"\\s*:\\s*\"user\"");
                            if (userMessageMatches.Count <= 1) {
                                isNewConversation = true;
                                Log("检测到单条用户消息，自动创建新对话");
                            }
                        } catch { }
                    }
                    
                    if (isNewConversation) {
                        Log("检测到新对话请求");
                    }
                    
                    var (prompt, images) = ExtractPromptAndImages(body);
                    if (string.IsNullOrWhiteSpace(prompt)) {
                        if (!body.Trim().StartsWith("{")) prompt = body;
                        else { ResponseText(ctx, "{\"error\": \"Parse Error\"}", 400, "application/json"); return; }
                    }
                    _currentContext = ctx;
                    _currentRequestId = Interlocked.Increment(ref _nextRequestId);
                    _currentRequestToken = Guid.NewGuid().ToString("N");
                    Interlocked.Exchange(ref _requestCompletionStarted, 0);
                    _isCurrentRequestStreaming = isStream;
                    _networkBuffer.Clear();
                    _plainResponseBuffer.Clear();
                    _lastEventType = "";
                    _hasSeenUserMessage = false;  // 重置标记
                    _sentImageUrls.Clear();  // 重置已处理图片集合

                    if (isStream) {
                        ctx.Response.ContentType = "text/event-stream";
                        ctx.Response.AppendHeader("Cache-Control", "no-cache");
                        ctx.Response.AppendHeader("Connection", "keep-alive");
                    } else { ctx.Response.ContentType = "application/json"; }

                    StartRequestTimeoutWatchdog(_currentRequestId, _currentRequestToken);
                    
                    BeginInvoke(new Action(() => _ = ExecuteRequestAsync(prompt, images, isNewConversation, _currentRequestId, _currentRequestToken)));
                }
            } catch (Exception ex) { Log($"Error: {ex.Message}"); if (_currentContext == ctx) CleanupRequest(); }
        } else { ResponseText(ctx, "Not Found", 404); }
    }

    private void StartRequestTimeoutWatchdog(int requestId, string requestToken)
    {
        Task.Run(async () => {
            await Task.Delay(RequestTimeout);
            if (requestId != _currentRequestId || _currentContext == null || requestToken != _currentRequestToken) return;
            if (Interlocked.CompareExchange(ref _requestCompletionStarted, 1, 1) != 0) return;
            try {
                if (!IsDisposed && IsHandleCreated) {
                    BeginInvoke(new Action(() => FailCurrentRequest(requestId, "请求超时，豆包页面没有在预期时间内返回结果。", 504)));
                }
            } catch { }
        });
    }

    private async Task ExecuteRequestAsync(string prompt, List<string> images, bool isNewConversation, int requestId, string requestToken)
    {
        try {
            if (requestId != _currentRequestId || _currentContext == null || requestToken != _currentRequestToken) return;

            if (images.Count > 0) {
                if (isNewConversation) {
                    bool ready = await EnsureNewConversationReady();
                    if (!ready) {
                        FailCurrentRequest(requestId, "新建对话失败，未能切换到可发送状态。");
                        return;
                    }
                }

                foreach (var img in images) {
                    await InjectImageToWebView(img, requestToken);
                }

                await Task.Delay(1000);
                bool injected = await InjectPromptToWebView(prompt, false, requestToken);
                if (!injected) {
                    FailCurrentRequest(requestId, "消息注入失败，未能定位输入框或触发发送。");
                }
                return;
            }

            bool sent = await InjectPromptToWebView(prompt, isNewConversation, requestToken);
            if (!sent) {
                FailCurrentRequest(requestId, "消息注入失败，未能定位输入框或触发发送。");
            }
        } catch (Exception ex) {
            Log($"请求执行失败: {ex.Message}");
            FailCurrentRequest(requestId, "请求执行失败: " + ex.Message);
        }
    }

    private (string text, List<string> images) ExtractPromptAndImages(string json)
    {
        var images = new List<string>();
        string textPrompt = "";

        try {
            var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

            // 尝试使用 Regex 解析 messages 数组
            // 这对于 Roo-Code 非常重要，因为它需要保留 system 消息中的工具指令
            var result = ExtractMessagesWithJson(json, seen, images);
            if (result.HasValue)
            {
                return result.Value;
            }

            // 如果 JSON 解析失败，回退到原有的 regex 方式
            Log("[DEBUG] JSON 解析失败，回退到 regex 方式");

            bool contentIsArray = false;
            string contentScope = json;

            // 优先抓取最后一条用户消息，兼容 content 为数组或纯字符串
            var userMatches = Regex.Matches(
                json,
                "\"role\"\\s*:\\s*\"user\".*?\"content\"\\s*:\\s*(\\[(.*?)\\]|\"((?:[^\"\\\\]|\\\\.)*)\")",
                RegexOptions.Singleline | RegexOptions.IgnoreCase
            );
            if (userMatches.Count > 0)
            {
                var lastUser = userMatches[userMatches.Count - 1];
                if (lastUser.Groups[2].Success && !string.IsNullOrWhiteSpace(lastUser.Groups[2].Value))
                {
                    contentScope = lastUser.Groups[2].Value; // content 为数组
                    contentIsArray = true;
                }
                else if (lastUser.Groups[3].Success)
                {
                    // content 为纯文本字符串
                    textPrompt = UnescapeJson(lastUser.Groups[3].Value);
                }
            }

            // 【重要修改】不再提前返回纯文本用户消息，而是继续提取 system 指令
            // 这对于 Roo-Code 非常重要，因为 system 消息包含工具协议

            // 只使用最后一个 content 数组，避免复用历史消息
            Match lastContentMatch = null;
            if (!contentIsArray)
            {
                var contentMatches = Regex.Matches(json, "\"content\"\\s*:\\s*\\[(.*?)\\]", RegexOptions.Singleline);
                if (contentMatches.Count > 0) lastContentMatch = contentMatches[contentMatches.Count - 1];
                contentScope = lastContentMatch != null ? lastContentMatch.Groups[1].Value : contentScope;
            }

            // 文本：抓取 content 列表中的 text，以及全局 text 字段作为兜底
            var textMatches = Regex.Matches(contentScope, "\"type\"\\s*:\\s*\"text\"[^}]*\"text\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"", RegexOptions.IgnoreCase);
            bool multipleTexts = textMatches.Count > 1;
            if (textMatches.Count > 0 && string.IsNullOrWhiteSpace(textPrompt))
            {
                // 只取最后一条文本，避免多轮历史堆叠
                var last = textMatches[textMatches.Count - 1];
                textPrompt = UnescapeJson(last.Groups[1].Value);
            }
            if (string.IsNullOrWhiteSpace(textPrompt))
            {
                var fallbackText = Regex.Matches(contentScope, "\"text\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"");
                if (fallbackText.Count > 0)
                {
                    var last = fallbackText[fallbackText.Count - 1];
                    textPrompt = UnescapeJson(last.Groups[1].Value);
                    multipleTexts = fallbackText.Count > 1;
                }
            }

            // 如果检测到多条文本（可能是历史堆叠），不携带附件，防止重复发送旧文件
            if (!multipleTexts)
            {
                // 附件/图片：尽可能多地抓取 url/file/data 三种形式
                string[] patterns = new [] {
                    "\"image_url\"[^}]*\"url\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"",
                    "\"file_url\"[^}]*\"url\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"",
                    "\"type\"\\s*:\\s*\"file\"[^}]*\"(?:url|data|base64)\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"",
                    "\"url\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*(?:\\.pptx|\\.ppt|\\.pdf|\\.docx|\\.doc|\\.xls|\\.xlsx)(?:[^\"\\\\]*))\""
                };
                foreach (var pat in patterns)
                {
                    var ms = Regex.Matches(contentScope, pat, RegexOptions.IgnoreCase);
                    foreach (Match m in ms)
                    {
                        string url = UnescapeJson(m.Groups[1].Value);
                        if (seen.Add(url)) { images.Add(url); LogAttachment(url); }
                    }
                }

                // data:...;base64,...
                var dataAll = Regex.Matches(contentScope, "data:[^\"\\\\\\s]+;base64,[^\"\\\\\\s]+", RegexOptions.IgnoreCase);
                foreach (Match m in dataAll)
                {
                    string url = UnescapeJson(m.Value);
                    if (seen.Add(url)) { images.Add(url); LogAttachment(url); }
                }

                // 本地路径
                var pathMatches = Regex.Matches(contentScope, "([A-Za-z]:\\\\[^\"\\s]+\\.(?:pptx|ppt|pdf|docx|doc|xls|xlsx))", RegexOptions.IgnoreCase);
                foreach (Match m in pathMatches)
                {
                    string path = UnescapeJson(m.Groups[1].Value);
                    if (seen.Add(path)) { images.Add(path); LogAttachment(path); }
                }
            }

            Log($"提取到 {images.Count} 个附件/图片");

            // 始终尝试提取 system 指令（对于 Roo-Code 非常重要）
            string finalPrompt = ExtractPromptFromJsonWithSystem(json, textPrompt);
            return (finalPrompt.Trim(), images);
        } catch (Exception ex) {
            Log($"解析错误: {ex.Message}");
            return (ExtractPromptFromJson(json), images);
        }
    }

    // 使用 Regex 解析 messages 数组（兼容 Quicker 环境，不依赖 System.Text.Json）
    private (string text, List<string> images)? ExtractMessagesWithJson(string json, HashSet<string> seen, List<string> images)
    {
        try {
            // 检查是否包含 messages 数组
            if (!json.Contains("\"messages\""))
            {
                return null;
            }

            StringBuilder systemPrompt = new StringBuilder();
            string lastUserContent = "";

            // 兼容 {"role": "...", "content": ...} 以及 {"content": ..., "role": "..."} 顺序，使用 [\s\S]*? 允许字段间有换行/其它字段
            var messagePattern = new Regex(
                "\"role\"\\s*:\\s*\"(system|user|assistant)\"[\\s\\S]*?\"content\"\\s*:\\s*(?:\"((?:[^\"\\\\]|\\\\.)*)\"|\\[(.*?)\\])|\"content\"\\s*:\\s*(?:\"((?:[^\"\\\\]|\\\\.)*)\"|\\[(.*?)\\])[\\s\\S]*?\"role\"\\s*:\\s*\"(system|user|assistant)\"",
                RegexOptions.IgnoreCase
            );

            var matches = messagePattern.Matches(json);
            foreach (Match match in matches)
            {
                // role 可能在前（Group1）也可能在后（Group6）
                string role = match.Groups[1].Success ? match.Groups[1].Value.ToLower() : match.Groups[6].Value.ToLower();
                // content 也分字符串/数组两种，兼容前后顺序
                string contentStr = match.Groups[2].Success ? match.Groups[2].Value : match.Groups[4].Value;
                string contentArray = match.Groups[3].Success ? match.Groups[3].Value : match.Groups[5].Value;

                string content = "";
                if (!string.IsNullOrEmpty(contentStr))
                {
                    // content 是字符串
                    content = UnescapeJson(contentStr);
                }
                else if (!string.IsNullOrEmpty(contentArray))
                {
                    // content 是数组，提取文本部分
                    content = ExtractTextFromContentArray(contentArray, seen, images);
                }

                if (role == "system")
                {
                    // 始终保留 system 消息（对于 Roo-Code 的工具指令非常重要）
                    if (!string.IsNullOrWhiteSpace(content))
                    {
                        if (systemPrompt.Length > 0) systemPrompt.Append("\n\n");
                        systemPrompt.Append(content);
                    }
                }
                else if (role == "user")
                {
                    // 只保留最后一条 user 消息
                    lastUserContent = content;
                }
                // assistant 消息通常不需要发送给豆包
            }

            // 构建最终提示词
            StringBuilder finalPrompt = new StringBuilder();

            // 始终添加 system 指令（这对 Roo-Code 非常关键）
            if (systemPrompt.Length > 0)
            {
                finalPrompt.Append("[System Instructions]:\n");
                finalPrompt.Append(systemPrompt.ToString());
                finalPrompt.Append("\n\n[User Message]:\n");
            }

            finalPrompt.Append(lastUserContent);

            Log($"[Regex解析] system长度: {systemPrompt.Length}, user长度: {lastUserContent.Length}, 附件: {images.Count}");

            return (finalPrompt.ToString().Trim(), images);
        }
        catch (Exception ex)
        {
            Log($"[Regex解析] 失败: {ex.Message}");
            return null;
        }
    }

    // 从 content 数组中提取文本内容（使用 Regex）
    private string ExtractTextFromContentArray(string contentArray, HashSet<string> seen, List<string> images)
    {
        StringBuilder textParts = new StringBuilder();

        // 提取 type: "text" 的文本
        var textPattern = new Regex(
            @"""type""\s*:\s*""text""[^}]*""text""\s*:\s*""((?:[^""\\]|\\.)*)""",
            RegexOptions.Singleline | RegexOptions.IgnoreCase
        );
        var textMatches = textPattern.Matches(contentArray);
        foreach (Match m in textMatches)
        {
            string text = UnescapeJson(m.Groups[1].Value);
            if (!string.IsNullOrWhiteSpace(text))
            {
                if (textParts.Length > 0) textParts.Append("\n");
                textParts.Append(text);
            }
        }

        // 提取图片 URL
        var imageUrlPattern = new Regex(
            @"""type""\s*:\s*""image(?:_url)?""\s*[^}]*""(?:image_)?url""\s*:\s*(?:\{[^}]*""url""\s*:\s*""((?:[^""\\]|\\.)*)""|""((?:[^""\\]|\\.)*)"")",
            RegexOptions.Singleline | RegexOptions.IgnoreCase
        );
        var imageMatches = imageUrlPattern.Matches(contentArray);
        foreach (Match m in imageMatches)
        {
            string url = m.Groups[1].Success ? m.Groups[1].Value : m.Groups[2].Value;
            url = UnescapeJson(url);
            if (!string.IsNullOrWhiteSpace(url) && seen.Add(url))
            {
                images.Add(url);
                LogAttachment(url);
            }
        }

        // 提取文件 URL
        var fileUrlPattern = new Regex(
            @"""type""\s*:\s*""file""\s*[^}]*""(?:file_)?url""\s*:\s*(?:\{[^}]*""url""\s*:\s*""((?:[^""\\]|\\.)*)""|""((?:[^""\\]|\\.)*)"")",
            RegexOptions.Singleline | RegexOptions.IgnoreCase
        );
        var fileMatches = fileUrlPattern.Matches(contentArray);
        foreach (Match m in fileMatches)
        {
            string url = m.Groups[1].Success ? m.Groups[1].Value : m.Groups[2].Value;
            url = UnescapeJson(url);
            if (!string.IsNullOrWhiteSpace(url) && seen.Add(url))
            {
                images.Add(url);
                LogAttachment(url);
            }
        }

        return textParts.ToString();
    }

    // 提取包含 system 指令的完整提示词
    private string ExtractPromptFromJsonWithSystem(string json, string userPrompt)
    {
        try {
            string systemPrompt = "";

            // 提取 system 消息
            var systemMatches = Regex.Matches(
                json,
                "\"role\"\\s*:\\s*\"system\".*?\"content\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"",
                RegexOptions.Singleline | RegexOptions.IgnoreCase
            );

            if (systemMatches.Count > 0)
            {
                StringBuilder sb = new StringBuilder();
                foreach (Match m in systemMatches)
                {
                    string content = UnescapeJson(m.Groups[1].Value);
                    if (!string.IsNullOrWhiteSpace(content))
                    {
                        if (sb.Length > 0) sb.Append("\n\n");
                        sb.Append(content);
                    }
                }
                systemPrompt = sb.ToString();
            }

            // 如果没有通过参数传入 userPrompt，尝试提取
            if (string.IsNullOrWhiteSpace(userPrompt))
            {
                userPrompt = ExtractPromptFromJson(json);
                // 如果 ExtractPromptFromJson 已经包含了 [System Instructions]，直接返回
                if (userPrompt.Contains("[System Instructions]:"))
                {
                    return userPrompt;
                }
            }

            // 构建最终提示词
            StringBuilder finalPrompt = new StringBuilder();
            if (!string.IsNullOrWhiteSpace(systemPrompt))
            {
                finalPrompt.Append("[System Instructions]:\n");
                finalPrompt.Append(systemPrompt);
                finalPrompt.Append("\n\n[User Message]:\n");
            }
            finalPrompt.Append(userPrompt);

            return finalPrompt.ToString().Trim();
        }
        catch (Exception ex)
        {
            Log($"提取 system 指令失败: {ex.Message}");
            return userPrompt;
        }
    }

    private string ExtractPromptFromJson(string json)
    {
        try {
            string systemPrompt = "", userPrompt = "";
            var roleRegex = new Regex("\"role\"\\s*:\\s*\"(system|user|assistant)\"");
            var contentRegex = new Regex("\"content\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"");
            var roles = roleRegex.Matches(json).Cast<Match>().ToList();
            var contents = contentRegex.Matches(json).Cast<Match>().ToList();

            if (roles.Count > 0 && roles.Count == contents.Count) {
                for (int i = 0; i < roles.Count; i++) {
                    string r = roles[i].Groups[1].Value.ToLower();
                    string c = UnescapeJson(contents[i].Groups[1].Value);
                    if (r == "system") systemPrompt += c + "\n\n";
                    else if (r == "user") userPrompt = c; 
                }
            } else {
                var lastMatch = contents.LastOrDefault();
                if (lastMatch != null) userPrompt = UnescapeJson(lastMatch.Groups[1].Value);
            }
            string finalPrompt = "";
            if (!string.IsNullOrWhiteSpace(systemPrompt)) finalPrompt += "[System Instructions]:\n" + systemPrompt.Trim() + "\n\n[User Message]:\n";
            finalPrompt += userPrompt;
            return finalPrompt.Trim();
        } catch { return null; }
    }

    private async Task InjectImageToWebView(string imageData, string requestToken = "")
    {
        try {
            string dataUrl = imageData;
            string fileName = "uploaded_file";

            if (imageData.StartsWith("data:", StringComparison.OrdinalIgnoreCase))
            {
                // base64 data, keep as-is
                try {
                    string mime = imageData.Substring(5, imageData.IndexOf(';') - 5);
                    string ext = GetExtensionFromContentType(mime);
                    if (!string.IsNullOrEmpty(ext)) fileName = "uploaded_file" + ext;
                } catch { }
            }
            else if (imageData.StartsWith("http://", StringComparison.OrdinalIgnoreCase) || imageData.StartsWith("https://", StringComparison.OrdinalIgnoreCase))
            {
                // 下载远程文件并转为 data URL
                using (var client = new WebClient())
                {
                    byte[] bytes = await client.DownloadDataTaskAsync(imageData);
                    string contentType = client.ResponseHeaders?["Content-Type"];
                    if (string.IsNullOrWhiteSpace(contentType)) contentType = "application/octet-stream";
                    fileName = Path.GetFileName(new Uri(imageData).AbsolutePath);
                    if (string.IsNullOrWhiteSpace(fileName)) fileName = "uploaded_file" + GetExtensionFromContentType(contentType);
                    else if (Path.GetExtension(fileName) == string.Empty) fileName = fileName + GetExtensionFromContentType(contentType);
                    dataUrl = BuildDataUrl(bytes, contentType);
                }
            }
            else if (imageData.StartsWith("file://", StringComparison.OrdinalIgnoreCase))
            {
                string localPath = new Uri(imageData).LocalPath;
                if (File.Exists(localPath))
                {
                    byte[] bytes = File.ReadAllBytes(localPath);
                    string contentType = GetContentTypeFromExtension(localPath);
                    fileName = Path.GetFileName(localPath);
                    dataUrl = BuildDataUrl(bytes, contentType);
                }
                else
                {
                    Log($"本地文件不存在: {localPath}");
                    return;
                }
            }
            else if (File.Exists(imageData))
            {
                // 本地文件路径
                byte[] bytes = File.ReadAllBytes(imageData);
                string contentType = GetContentTypeFromExtension(imageData);
                fileName = Path.GetFileName(imageData);
                if (Path.GetExtension(fileName) == string.Empty) fileName = fileName + GetExtensionFromContentType(contentType);
                dataUrl = BuildDataUrl(bytes, contentType);
            }
            else
            {
                Log($"不支持的图片/文件格式: {imageData.Substring(0, Math.Min(80, imageData.Length))}");
                return;
            }

            string escapedData = dataUrl.Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\n", "").Replace("\r", "");
            string escapedFile = fileName.Replace("\\", "").Replace("\"", "");
            string script = $"window.__doubaoRelayActiveRequestToken=\"{EscapeForJsString(requestToken)}\"; window.DoubaoBridge.uploadImage(\"{escapedData}\", \"{escapedFile}\")";
            Log("正在上传图片/文件...");
            if (_coreWebView != null) await _coreWebView.ExecuteScriptAsync(script);
        } catch (Exception ex) {
            Log($"图片上传失败: {ex.Message}");
        }
    }

    private string BuildDataUrl(byte[] bytes, string contentType)
    {
        string base64 = Convert.ToBase64String(bytes);
        return $"data:{contentType};base64,{base64}";
    }

    private string GetExtensionFromContentType(string contentType)
    {
        if (string.IsNullOrWhiteSpace(contentType)) return "";
        switch (contentType.ToLowerInvariant())
        {
            case "image/png": return ".png";
            case "image/jpeg": return ".jpg";
            case "image/jpg": return ".jpg";
            case "image/gif": return ".gif";
            case "image/bmp": return ".bmp";
            case "image/webp": return ".webp";
            case "image/heic": return ".heic";
            case "image/svg+xml": return ".svg";
            case "application/pdf": return ".pdf";
            case "application/vnd.ms-powerpoint": return ".ppt";
            case "application/vnd.openxmlformats-officedocument.presentationml.presentation": return ".pptx";
            case "application/msword": return ".doc";
            case "application/vnd.openxmlformats-officedocument.wordprocessingml.document": return ".docx";
            case "application/vnd.ms-excel": return ".xls";
            case "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet": return ".xlsx";
            default: return "";
        }
    }

    private string GetContentTypeFromExtension(string path)
    {
        string ext = Path.GetExtension(path)?.ToLowerInvariant() ?? "";
        switch (ext)
        {
            case ".png": return "image/png";
            case ".jpg":
            case ".jpeg": return "image/jpeg";
            case ".gif": return "image/gif";
            case ".bmp": return "image/bmp";
            case ".webp": return "image/webp";
            case ".heic": return "image/heic";
            case ".svg": return "image/svg+xml";
            case ".pdf": return "application/pdf";
            case ".ppt":
            case ".pptx": return "application/vnd.openxmlformats-officedocument.presentationml.presentation";
            case ".doc":
            case ".docx": return "application/vnd.openxmlformats-officedocument.wordprocessingml.document";
            case ".xls":
            case ".xlsx": return "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet";
            case ".txt": return "text/plain";
            default: return "application/octet-stream";
        }
    }

    private void OnWebMessageReceived(object sender, CoreWebView2WebMessageReceivedEventArgs e)
    {
        try {
            string msg = e.TryGetWebMessageAsString();
            Log($"[DEBUG] 收到消息: {msg.Substring(0, Math.Min(50, msg.Length))}...");
            
            if (msg.StartsWith("[LOG]")) { Log(msg); return; }
            
            if (TryParseRelayMessage(msg, "[GENERATED_IMAGE]|", out var imageToken, out var imageUrl)) {
                if (!IsCurrentRequestToken(imageToken)) {
                    Log($"[DEBUG] 忽略过期图片消息 token={imageToken}");
                    return;
                }
                HandleGeneratedImageUrl(imageUrl);
                return;
            }

            if (msg.StartsWith("[GENERATED_IMAGE]")) {
                string legacyImageUrl = msg.Substring(17);
                HandleGeneratedImageUrl(legacyImageUrl);
                return;
            }
            
            if (_currentContext == null) {
                Log($"[DEBUG] 当前无活动请求，忽略消息: {msg.Substring(0, Math.Min(50, msg.Length))}...");
                return;
            }
            
            if (TryParseRelayMessage(msg, "[NETWORK_DONE]|", out var doneToken, out _)) {
                if (!IsCurrentRequestToken(doneToken)) {
                    Log($"[DEBUG] 忽略过期NETWORK_DONE token={doneToken}");
                    return;
                }
                Log("[DEBUG] 处理NETWORK_DONE消息");
                _ = FinishRequestAsync(_currentRequestId);
                return;
            }

            if (msg.StartsWith("[NETWORK_DONE]")) {
                Log("[DEBUG] 处理旧格式NETWORK_DONE消息");
                _ = FinishRequestAsync(_currentRequestId);
                return;
            }
            
            if (TryParseRelayMessage(msg, "[NETWORK_DATA]|", out var dataToken, out var rawData)) {
                if (!IsCurrentRequestToken(dataToken)) {
                    Log($"[DEBUG] 忽略过期NETWORK_DATA token={dataToken}");
                    return;
                }
                Log($"[DEBUG] 处理NETWORK_DATA消息，长度: {rawData.Length}");
                ProcessNetworkData(rawData);
                return;
            }

            if (msg.StartsWith("[NETWORK_DATA]")) {
                string legacyRawData = msg.Substring(14);
                Log($"[DEBUG] 处理旧格式NETWORK_DATA消息，长度: {legacyRawData.Length}");
                ProcessNetworkData(legacyRawData);
            }
        } catch (Exception ex) { Log($"消息处理错误: {ex.Message}"); }
    }

    private void HandleGeneratedImageUrl(string imageUrl)
    {
        imageUrl = NormalizeImageUrl(imageUrl);
        if (string.IsNullOrWhiteSpace(imageUrl)) return;
        
        if (!IsAllowedImageUrl(imageUrl)) {
            Log($"[IMAGE_DEBUG] 跳过不需要的图片URL: {imageUrl.Substring(0, Math.Min(80, imageUrl.Length))}");
            return;
        }

        if (_sentImageUrls.Contains(imageUrl)) {
            Log($"[IMAGE_DEBUG] 图片URL已处理，跳过: {imageUrl.Substring(0, Math.Min(80, imageUrl.Length))}");
            return;
        }
        _sentImageUrls.Add(imageUrl);
        _isImageGenerationPending = false;  // 已经拿到图片URL，不再等待提示
        Log($"[IMAGE] ✅ 拦截到图片URL: {imageUrl}");

        if (_currentContext != null && !_hasProcessedImage) {
            _hasProcessedImage = true;
            SendDelta($"\n\n![generated_image]({imageUrl})\n\n");
            Log($"[NET] \\n\\n![generated_image]({imageUrl})\\n\\n");
        }
        else if (_currentContext != null) {
            _pendingImageUrls.Add(imageUrl);
            Log($"[IMAGE] 图片已处理，添加到待处理列表: {imageUrl.Substring(0, Math.Min(50, imageUrl.Length))}...");
        }
        else {
            _pendingImageUrls.Add(imageUrl);
            Log($"[IMAGE] 请求已结束，存储图片URL待处理: {imageUrl.Substring(0, Math.Min(50, imageUrl.Length))}...");
        }
    }

    private IEnumerable<string> ExtractImageUrlsFromJson(string json)
    {
        if (string.IsNullOrWhiteSpace(json)) yield break;
        // 同时匹配带转义斜杠和普通URL
        var matches = Regex.Matches(json, @"https?:\\/\\/[^""]*rc_gen_image[^""]*|https?://[^""\s]*rc_gen_image[^""\s]*", RegexOptions.IgnoreCase);
        foreach (Match m in matches) {
            string url = NormalizeImageUrl(m.Value);
            if (string.IsNullOrWhiteSpace(url)) continue;
            if (!IsAllowedImageUrl(url)) continue;
            yield return url;
        }
    }
    private void ProcessNetworkData(string rawChunk)
    {
        _networkBuffer.Append(rawChunk);
        string bufferStr = _networkBuffer.ToString();
        
        if (_isDebugMode && rawChunk.Length > 0) { 
             string debugSnippet = rawChunk.Length > 100 ? rawChunk.Substring(0, 100) + "..." : rawChunk;
             Log($"[RAW] {debugSnippet.Replace("\n", "\\n")}");
        }
        
        int lastNewLine = bufferStr.LastIndexOf('\n');
        if (lastNewLine == -1) return; 
        
        string processablePart = bufferStr.Substring(0, lastNewLine + 1);
        _networkBuffer.Remove(0, lastNewLine + 1);

        using (StringReader reader = new StringReader(processablePart))
        {
            string line;
            while ((line = reader.ReadLine()) != null) {
                if (string.IsNullOrWhiteSpace(line)) { _lastEventType = ""; continue; }
                line = line.Trim();
                
                if (line.StartsWith("event:")) { 
                    _lastEventType = line.Substring(6).Trim(); 
                }
                else if (line.StartsWith("data:")) {
                    string json = line.Substring(5).Trim();
                    
                    // 从豆包的SSE data中提取文本内容
                    string content = ExtractTextFromDoubaoData(json);
                    
                    if (!string.IsNullOrEmpty(content) && content != "[DONE]") {
                        SendDelta(content);
                    }
                }
            }
        }
    }

    private string ExtractTextFromDoubaoData(string json)
    {
        // 先尝试从JSON中提取图片URL
        foreach (var imgUrl in ExtractImageUrlsFromJson(json)) {
            HandleGeneratedImageUrl(imgUrl);
        }
        
        // 调试日志
        if (_isDebugMode && json.Length < 500) {
            Log($"[DEBUG_JSON] {json}");
        } else if (_isDebugMode && json.Length >= 500) {
            Log($"[DEBUG_JSON] {json.Substring(0, 500)}...");
        }

        // 1. 处理总结信息
        if (json.Contains("\"brief\":"))
        {
            var briefMatch = Regex.Match(json, "\"brief\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"");
            if (briefMatch.Success)
            {
                string brief = UnescapeJson(briefMatch.Groups[1].Value);
                Log($"[INFO] 最终摘要: {brief}");
            }
        }
        
        // 2. 更新块 ID 和 状态切换逻辑
        // 检测思考父块 (block_type: 10040)
        if (json.Contains("\"block_type\":10040") || json.Contains("\"block_type\": 10040"))
        {
            var idMatch = Regex.Match(json, "\"block_id\"\\s*:\\s*\"([^\"]*)\"");
            if (idMatch.Success)
            {
                _thinkingBlockId = idMatch.Groups[1].Value;
                Log($"[THINK] 更新思考父块 ID: {_thinkingBlockId}");
            }
        }

        // 仅在检测到显式的新块定义 (block_type: 10000) 时更新持久化的思考状态
        // 这样可以确保中间的 CHUNK_DELTA (通常不带块信息) 能够继承正确的状态
        if (json.Contains("\"block_type\":10000") || json.Contains("\"block_type\": 10000"))
        {
            // 跳过用户问题块
            if (!_hasSeenUserMessage)
            {
                _hasSeenUserMessage = true;
                Log("[SKIP] 跳过用户问题块");
                return null;
            }

            // 检查这个新块是否为思考块的子块
            if (!string.IsNullOrEmpty(_thinkingBlockId) && json.Contains($"\"parent_id\":\"{_thinkingBlockId}\""))
            {
                if (!_isCurrentBlockThinking) {
                    _isCurrentBlockThinking = true;
                    Log("[THINK] 进入思考模式");
                }
            }
            else
            {
                // 如果是 10000 块但没有 parent_id，或者是其他 parent，则视为正式回答的文字
                if (_isCurrentBlockThinking) {
                    _isCurrentBlockThinking = false;
                    Log("[THINK] 退出思考模式 (检测到新的独立块或非思考 Parent)");
                }
            }
        }
        
        // 处理图片提示过滤
        if (_hasProcessedImage && (json.Contains("以下是为你生成的图片") || json.Contains("为您生成图片") || json.Contains("生成的图片")))
        {
            return null;
        }
        if (!_hasProcessedImage && (json.Contains("以下是为你生成的图片") || json.Contains("为您生成图片") || json.Contains("生成的图片")))
        {
            _isImageGenerationPending = true;
            return "正在生成图片，请稍候...\n\n";
        }

        // 3. 提取文字内容
        string content = null;

        // 优先提取 text_block (STREAM_MSG_NOTIFY/PATCH)
        var textMatch = Regex.Match(json, "\"text_block\"\\s*:\\s*\\{\\s*\"text\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"");
        if (textMatch.Success)
        {
            content = UnescapeJson(textMatch.Groups[1].Value);
        }
        // 兜底提取 text 字段 (CHUNK_DELTA/CHUNK_DELTA)
        else
        {
            var deltaMatch = Regex.Match(json, "\"text\"\\s*:\\s*\"((?:[^\"\\\\]|\\\\.)*)\"");
            if (deltaMatch.Success)
            {
                content = UnescapeJson(deltaMatch.Groups[1].Value);
            }
        }

        // 4. 根据当前持久化的模式组装返回内容
        if (!string.IsNullOrEmpty(content))
        {
            if (_isCurrentBlockThinking)
            {
                if (!_isThinkingActive)
                {
                    _isThinkingActive = true;
                    return "<think>\n" + content;
                }
                return content;
            }
            else
            {
                if (_isThinkingActive)
                {
                    _isThinkingActive = false;
                    return "\n</think>\n\n" + content;
                }
                return content;
            }
        }

        return null;
    }

    private string UnescapeJson(string str) {
        if (string.IsNullOrEmpty(str)) return str;

        // 先处理 Unicode 转义序列 \uXXXX（这对于 Roo-Code 的 XML 标签非常重要）
        // 例如 \u003c 需要转换为 < ，\u003e 需要转换为 >
        str = Regex.Replace(str, @"\\u([0-9A-Fa-f]{4})", match => {
            int code = int.Parse(match.Groups[1].Value, System.Globalization.NumberStyles.HexNumber);
            return ((char)code).ToString();
        });

        // 然后处理标准 JSON 转义
        return str.Replace("\\\"", "\"").Replace("\\\\", "\\").Replace("\\n", "\n").Replace("\\r", "\r").Replace("\\t", "\t");
    }

    private void SendDelta(string text) {
        if (_isCurrentRequestStreaming) {
            string jsonChunk = $"{{\"id\":\"chatcmpl-doubao\",\"object\":\"chat.completion.chunk\",\"created\":{DateTimeOffset.Now.ToUnixTimeSeconds()},\"model\":\"doubao-chat\",\"choices\":[{{\"index\":0,\"delta\":{{\"content\":\"{EscapeJson(text)}\"}},\"finish_reason\":null}}]}}";
            WriteSseData($"data: {jsonChunk}\n\n");
        } else {
            _plainResponseBuffer.Append(text);
        }
        string logText = text.Replace("\n", "\\n");
        Log($"[NET] {logText}");
    }

    private async Task FinishRequestAsync(int requestId) {
        if (!TryBeginRequestCompletion(requestId)) return;
        Log("Stream Finished");
        
        // 如果结束时还在思考，补全标签
        if (_isThinkingActive)
        {
            SendDelta("\n</think>\n\n");
            _isThinkingActive = false;
        }

        if (_networkBuffer.Length > 0) { ProcessNetworkData("\n"); _networkBuffer.Clear(); }
        
        // 如果检测到图片生成，等待一段时间以便接收图片URL
        if (_hasProcessedImage || _pendingImageUrls.Count > 0 || _isImageGenerationPending) {
            Log("[IMAGE] 检测到图片生成，等待图片URL...");
            
            // 保存当前上下文引用，防止在等待期间被清理
            var currentContext = _currentContext;
            
            // 等待更长时间，确保图片URL能够被接收
            for (int i = 0; i < 10; i++) { // 最多等待10秒，每秒检查一次
                await Task.Delay(1000);
                
                if (_pendingImageUrls.Count > 0 && currentContext != null) {
                    Log($"[IMAGE] 发送 {_pendingImageUrls.Count} 个待处理的图片URL");
                    foreach (var imageUrl in _pendingImageUrls) {
                        try {
                            if (_isCurrentRequestStreaming) {
                                string jsonChunk = $"{{\"id\":\"chatcmpl-doubao\",\"object\":\"chat.completion.chunk\",\"created\":{DateTimeOffset.Now.ToUnixTimeSeconds()},\"model\":\"doubao-chat\",\"choices\":[{{\"index\":0,\"delta\":{{\"content\":\"\\n\\n![generated_image]({EscapeJson(imageUrl)})\\n\\n\"}},\"finish_reason\":null}}]}}";
                                string sseData = $"data: {jsonChunk}\n\n";
                                byte[] buffer = Encoding.UTF8.GetBytes(sseData);
                                currentContext.Response.OutputStream.Write(buffer, 0, buffer.Length);
                                currentContext.Response.OutputStream.Flush();
                            } else {
                                _plainResponseBuffer.Append($"\n\n![generated_image]({imageUrl})\n\n");
                            }
                            Log($"[NET] \\n\\n![generated_image]({imageUrl})\\n\\n");
                        } catch (Exception ex) {
                            Log($"[IMAGE] 发送图片URL失败: {ex.Message}");
                        }
                    }
                    _pendingImageUrls.Clear();
                }
                
                // 如果已经等待了5秒还没有新的图片URL，提前退出
                if (i >= 5 && _pendingImageUrls.Count == 0 && !_isImageGenerationPending) {
                    Log("[IMAGE] 5秒内没有新的图片URL，提前退出等待");
                    break;
                }
            }
            _isImageGenerationPending = false;
        }
        
        if (_isCurrentRequestStreaming) {
            string stopChunk = $"{{\"id\":\"chatcmpl-doubao\",\"object\":\"chat.completion.chunk\",\"created\":{DateTimeOffset.Now.ToUnixTimeSeconds()},\"model\":\"doubao-chat\",\"choices\":[{{\"index\":0,\"delta\":{{}},\"finish_reason\":\"stop\"}}]}}";
            WriteSseData($"data: {stopChunk}\n\n");
            WriteSseData("data: [DONE]\n\n");
        } else {
            string content = EscapeJson(_plainResponseBuffer.ToString());
            string finalJson = $"{{\"id\":\"chatcmpl-doubao\",\"object\":\"chat.completion\",\"created\":{DateTimeOffset.Now.ToUnixTimeSeconds()},\"model\":\"doubao-chat\",\"choices\":[{{\"index\":0,\"finish_reason\":\"stop\",\"message\":{{\"role\":\"assistant\",\"content\":\"{content}\"}}}}]}}";
            ResponseText(_currentContext, finalJson, 200, "application/json");
        }
        CleanupRequest();
    }

    private bool TryBeginRequestCompletion(int requestId)
    {
        if (requestId == 0 || requestId != _currentRequestId || _currentContext == null) return false;
        return Interlocked.CompareExchange(ref _requestCompletionStarted, 1, 0) == 0;
    }

    private void FailCurrentRequest(int requestId, string message, int statusCode = 500)
    {
        if (!TryBeginRequestCompletion(requestId)) return;
        Log("[ERROR] " + message);

        if (_isCurrentRequestStreaming) {
            string errorChunk = $"{{\"id\":\"chatcmpl-doubao\",\"object\":\"chat.completion.chunk\",\"created\":{DateTimeOffset.Now.ToUnixTimeSeconds()},\"model\":\"doubao-chat\",\"choices\":[{{\"index\":0,\"delta\":{{}},\"finish_reason\":\"error\"}}],\"error\":{{\"message\":\"{EscapeJson(message)}\",\"type\":\"relay_error\"}}}}";
            WriteSseData($"data: {errorChunk}\n\n");
            WriteSseData("data: [DONE]\n\n");
        } else if (_currentContext != null) {
            string errorJson = $"{{\"error\":{{\"message\":\"{EscapeJson(message)}\",\"type\":\"relay_error\"}}}}";
            ResponseText(_currentContext, errorJson, statusCode, "application/json");
        }

        CleanupRequest();
    }

    private void WriteSseData(string data) {
        try {
            byte[] buffer = Encoding.UTF8.GetBytes(data);
            _currentContext.Response.OutputStream.Write(buffer, 0, buffer.Length);
            _currentContext.Response.OutputStream.Flush();
        } catch { CleanupRequest(); }
    }

    private void CleanupRequest() {
        try {
            _currentContext?.Response.Close();
        } catch { }
        _currentContext = null;
        _currentRequestId = 0;
        _currentRequestToken = "";
        Interlocked.Exchange(ref _requestCompletionStarted, 0);
        _networkBuffer.Clear();
        _plainResponseBuffer.Clear();
        _thinkingBlockId = ""; // 重置思考块ID
        _isThinkingActive = false; // 重置思考激活状态
        _isCurrentBlockThinking = false; // 重置当前块思考状态
        _hasSeenUserMessage = false;  // 重置标记
        _hasProcessedImage = false;  // 重置图片处理标记
        _isImageGenerationPending = false;  // 重置等待图片标记
        _pendingImageUrls.Clear();  // 清空待处理的图片URL
        _sentImageUrls.Clear();  // 清空已发送图片URL记录
    }

    private void ResponseText(HttpListenerContext ctx, string text, int code = 200, string contentType = "text/plain") {
        try {
            ctx.Response.StatusCode = code; ctx.Response.ContentType = contentType;
            byte[] buffer = Encoding.UTF8.GetBytes(text);
            ctx.Response.ContentLength64 = buffer.Length;
            ctx.Response.OutputStream.Write(buffer, 0, buffer.Length);
            ctx.Response.Close();
        } catch { }
    }

    private bool IsCurrentRequestToken(string token)
    {
        return !string.IsNullOrWhiteSpace(token) && string.Equals(token, _currentRequestToken, StringComparison.Ordinal);
    }

    private bool TryParseRelayMessage(string msg, string prefix, out string token, out string payload)
    {
        token = "";
        payload = "";
        if (!msg.StartsWith(prefix)) return false;
        int tokenStart = prefix.Length;
        int payloadStart = msg.IndexOf('|', tokenStart);
        if (payloadStart < 0) {
            token = msg.Substring(tokenStart);
            return true;
        }
        token = msg.Substring(tokenStart, payloadStart - tokenStart);
        payload = msg.Substring(payloadStart + 1);
        return true;
    }

    private async Task<bool> ExecuteBoolScriptAsync(string script)
    {
        if (_coreWebView == null) return false;
        string result = await _coreWebView.ExecuteScriptAsync(script);
        string normalized = (result ?? "").Trim();
        if (string.Equals(normalized, "true", StringComparison.OrdinalIgnoreCase)) return true;
        if (string.Equals(normalized, "\"true\"", StringComparison.OrdinalIgnoreCase)) return true;
        return false;
    }

    private async Task<bool> ExecuteScriptWithFallbackAsync(string script)
    {
        if (_coreWebView == null) return false;
        string result = await _coreWebView.ExecuteScriptAsync(script);
        string normalized = (result ?? "").Trim();

        if (string.Equals(normalized, "false", StringComparison.OrdinalIgnoreCase)
            || string.Equals(normalized, "\"false\"", StringComparison.OrdinalIgnoreCase)) {
            return false;
        }

        // sendPrompt 的真实发送是异步完成的；只要脚本成功执行且没有显式 false，就继续等待网络回包。
        return true;
    }

    private bool IsAllowedImageUrl(string url)
    {
        if (string.IsNullOrWhiteSpace(url)) return false;
        if (!Uri.TryCreate(url, UriKind.Absolute, out var uri)) return false;

        if (uri.Host.IndexOf("byteimg.com", StringComparison.OrdinalIgnoreCase) < 0) return false;

        if (uri.AbsolutePath.IndexOf("rc_gen_image", StringComparison.OrdinalIgnoreCase) < 0) return false;
        return true;
    }

        script.AppendLine("                        tag === 'button' ||");
        script.AppendLine("                        tag === 'a' ||");
        script.AppendLine("                        role === 'button' ||");
        script.AppendLine("                        cls.includes('cursor-pointer') ||");
        script.AppendLine("                        cls.includes('sidebar_nav_item')");
        script.AppendLine("                    ) {");
        script.AppendLine("                        window.chrome.webview.postMessage('[LOG] 通过文本节点找到并点击');");
        script.AppendLine("                        el.click();");
        script.AppendLine("                        await new Promise(r => setTimeout(r, 1500));");
        script.AppendLine("                        window.chrome.webview.postMessage('[LOG] 新建对话完成');");
        script.AppendLine("                        return true;");
        script.AppendLine("                    }");
        script.AppendLine("");
        script.AppendLine("                    el = el.parentElement;");
        script.AppendLine("                }");
        script.AppendLine("            }");

        // 方式3：都找不到就强制刷新聊天页
        script.AppendLine("            window.chrome.webview.postMessage('[LOG] 未找到新对话按钮，强制刷新聊天页');");
        script.AppendLine("            window.location.href = 'https://www.doubao.com/chat/?t=' + Date.now();");
        script.AppendLine("            await new Promise(r => setTimeout(r, 2500));");
        script.AppendLine("            return true;");
        script.AppendLine("        } catch (e) {");
        script.AppendLine("            window.chrome.webview.postMessage('[LOG] 新建对话失败: ' + e.message);");
        script.AppendLine("            return false;");
        script.AppendLine("        }");
        script.AppendLine("    },");
        
        // sendPrompt function
        script.AppendLine("    sendPrompt: async function(text) {");
        script.AppendLine("        const isVisible = (el) => {");
        script.AppendLine("            if (!el) return false;");
        script.AppendLine("            const rect = el.getBoundingClientRect();");
        script.AppendLine("            const style = window.getComputedStyle(el);");
        script.AppendLine("            return rect.width > 0 && rect.height > 0 && style.visibility !== 'hidden' && style.display !== 'none';");
        script.AppendLine("        };");
        script.AppendLine("        const findInput = () => {");
        script.AppendLine("            const selectors = [");
        script.AppendLine("                'textarea.semi-input-textarea',");
        script.AppendLine("                'textarea[placeholder*=\"发消息\"]',");
        script.AppendLine("                'textarea[placeholder*=\"说话\"]',");
        script.AppendLine("                'textarea[placeholder*=\"输入\"]',");
        script.AppendLine("                'textarea[placeholder*=\"问\"]',");
        script.AppendLine("                'textarea',");
        script.AppendLine("                '[contenteditable=\"true\"][role=\"textbox\"]',");
        script.AppendLine("                '[contenteditable=\"true\"]'");
        script.AppendLine("            ];");
        script.AppendLine("            const candidates = [];");
        script.AppendLine("            for (const selector of selectors) {");
        script.AppendLine("                for (const el of document.querySelectorAll(selector)) {");
        script.AppendLine("                    if (!candidates.includes(el)) candidates.push(el);");
        script.AppendLine("                }");
        script.AppendLine("            }");
        script.AppendLine("            const usable = candidates.filter(el => isVisible(el) && !el.disabled && !el.readOnly);");
        script.AppendLine("            usable.sort((a, b) => {");
        script.AppendLine("                const score = (el) => {");
        script.AppendLine("                    const rect = el.getBoundingClientRect();");
        script.AppendLine("                    const placeholder = (el.getAttribute('placeholder') || '').toLowerCase();");
        script.AppendLine("                    const cls = (el.className || '').toString().toLowerCase();");
        script.AppendLine("                    let value = rect.bottom;");
        script.AppendLine("                    if (cls.includes('semi-input-textarea')) value += 1000;");
        script.AppendLine("                    if (placeholder.includes('发消息') || placeholder.includes('输入') || placeholder.includes('问')) value += 800;");
        script.AppendLine("                    if (el.tagName === 'TEXTAREA') value += 300;");
        script.AppendLine("                    return value;");
        script.AppendLine("                };");
        script.AppendLine("                return score(b) - score(a);");
        script.AppendLine("            });");
        script.AppendLine("            return usable[0] || null;");
        script.AppendLine("        };");
        script.AppendLine("        const waitForInput = async (timeoutMs = 10000) => {");
        script.AppendLine("            const started = Date.now();");
        script.AppendLine("            while (Date.now() - started < timeoutMs) {");
        script.AppendLine("                const input = findInput();");
        script.AppendLine("                if (input) return input;");
        script.AppendLine("                await new Promise(resolve => setTimeout(resolve, 250));");
        script.AppendLine("            }");
        script.AppendLine("            return null;");
        script.AppendLine("        };");
        script.AppendLine("        const setInputValue = (input, value) => {");
        script.AppendLine("            input.focus();");
        script.AppendLine("            const tag = input.tagName;");
        script.AppendLine("            if (tag === 'TEXTAREA' || tag === 'INPUT') {");
        script.AppendLine("                const proto = tag === 'TEXTAREA' ? window.HTMLTextAreaElement.prototype : window.HTMLInputElement.prototype;");
        script.AppendLine("                const descriptor = Object.getOwnPropertyDescriptor(proto, 'value');");
        script.AppendLine("                if (descriptor && descriptor.set) descriptor.set.call(input, value);");
        script.AppendLine("                else input.value = value;");
        script.AppendLine("                try { input.dispatchEvent(new InputEvent('input', { bubbles: true, inputType: 'insertText', data: value })); }");
        script.AppendLine("                catch (e) { input.dispatchEvent(new Event('input', { bubbles: true })); }");
        script.AppendLine("                input.dispatchEvent(new Event('change', { bubbles: true }));");
        script.AppendLine("            } else {");
        script.AppendLine("                input.textContent = value;");
        script.AppendLine("                try { input.dispatchEvent(new InputEvent('input', { bubbles: true, inputType: 'insertText', data: value })); }");
        script.AppendLine("                catch (e) { input.dispatchEvent(new Event('input', { bubbles: true })); }");
        script.AppendLine("            }");
        script.AppendLine("        };");
        script.AppendLine("        const readInputValue = (input) => {");
        script.AppendLine("            if (!input) return '';");
        script.AppendLine("            return input.value !== undefined ? input.value : (input.textContent || '');");
        script.AppendLine("        };");
        script.AppendLine("        const input = await waitForInput();");
        script.AppendLine("        if (!input) {");
        script.AppendLine("            window.chrome.webview.postMessage('[LOG] 错误: 未找到输入框，当前URL: ' + location.href);");
        script.AppendLine("            window.chrome.webview.postMessage('[LOG] 输入框候选数量 textarea=' + document.querySelectorAll('textarea').length + ', editable=' + document.querySelectorAll('[contenteditable=\"true\"]').length);");
        script.AppendLine("            return false;");
        script.AppendLine("        }");
        script.AppendLine("        window.chrome.webview.postMessage('[LOG] 找到输入框: ' + input.tagName + ' ' + (input.className || '') + ' ' + (input.getAttribute('placeholder') || ''));");
        script.AppendLine("        setInputValue(input, text);");
        script.AppendLine("        input.focus();");
        script.AppendLine("        const findSendButton = () => {");
        script.AppendLine("            const roots = [];");
        script.AppendLine("            let cursor = input;");
        script.AppendLine("            for (let i = 0; cursor && i < 6; i++, cursor = cursor.parentElement) roots.push(cursor);");
        script.AppendLine("            roots.push(document);");
        script.AppendLine("            const candidates = [];");
        script.AppendLine("            for (const root of roots) {");
        script.AppendLine("                const buttons = root.querySelectorAll ? root.querySelectorAll('button, [role=\"button\"]') : [];");
        script.AppendLine("                for (const btn of buttons) if (!candidates.includes(btn)) candidates.push(btn);");
        script.AppendLine("            }");
        script.AppendLine("            const visible = candidates.filter(btn => isVisible(btn) && !btn.disabled && btn.getAttribute('aria-disabled') !== 'true' && btn.getAttribute('data-disabled') !== 'true');");
        script.AppendLine("            const inputRect = input.getBoundingClientRect();");
        script.AppendLine("            const scored = visible.map(btn => {");
        script.AppendLine("                const rect = btn.getBoundingClientRect();");
        script.AppendLine("                const textValue = ((btn.textContent || '') + ' ' + (btn.getAttribute('aria-label') || '') + ' ' + (btn.getAttribute('title') || '')).toLowerCase();");
        script.AppendLine("                let score = 0;");
        script.AppendLine("                if (btn.type === 'submit') score += 1000;");
        script.AppendLine("                if (textValue.includes('发送') || textValue.includes('send')) score += 1000;");
        script.AppendLine("                if (rect.left >= inputRect.left && rect.top >= inputRect.top - 80) score += 250;");
        script.AppendLine("                if (rect.right >= inputRect.right - 160) score += 200;");
        script.AppendLine("                score -= Math.abs(rect.top - inputRect.top) + Math.abs(rect.left - inputRect.right) / 3;");
        script.AppendLine("                return { btn, score };");
        script.AppendLine("            }).filter(item => item.score > -200);");
        script.AppendLine("            scored.sort((a, b) => b.score - a.score);");
        script.AppendLine("            return scored.length > 0 ? scored[0].btn : null;");
        script.AppendLine("        };");
        script.AppendLine("        const clickButton = (retryCount = 0) => {");
        script.AppendLine("            const maxRetries = 3;");
        script.AppendLine("            const sendBtn = findSendButton();");
        script.AppendLine("            if (sendBtn) {");
        script.AppendLine("                window.chrome.webview.postMessage('[LOG] 点击发送按钮');");
        script.AppendLine("                sendBtn.click();");
        script.AppendLine("                setTimeout(() => {");
        script.AppendLine("                    if (readInputValue(input).length > 0 && retryCount < maxRetries) {");
        script.AppendLine("                        window.chrome.webview.postMessage('[LOG] 发送失败，500ms后重试...');");
        script.AppendLine("                        setTimeout(() => clickButton(retryCount + 1), 500);");
        script.AppendLine("                    } else if (readInputValue(input).length === 0) {");
        script.AppendLine("                        window.chrome.webview.postMessage('[LOG] ✅ 发送成功！');");
        script.AppendLine("                    } else {");
        script.AppendLine("                        window.chrome.webview.postMessage('[LOG] ⚠️ 多次重试后仍失败');");
        script.AppendLine("                    }");
        script.AppendLine("                }, 300);");
        script.AppendLine("            } else {");
        script.AppendLine("                if (retryCount < maxRetries) {");
        script.AppendLine("                    window.chrome.webview.postMessage('[LOG] 未找到按钮，500ms后重试...');");
        script.AppendLine("                    setTimeout(() => clickButton(retryCount + 1), 500);");
        script.AppendLine("                } else {");
        script.AppendLine("                    window.chrome.webview.postMessage('[LOG] ❌ 找不到发送按钮');");
        script.AppendLine("                }");
        script.AppendLine("            }");
        script.AppendLine("        };");
        script.AppendLine("        const pressEnter = () => {");
        script.AppendLine("            const eventInit = { key: 'Enter', code: 'Enter', keyCode: 13, which: 13, bubbles: true, cancelable: true };");
        script.AppendLine("            input.dispatchEvent(new KeyboardEvent('keydown', eventInit));");
        script.AppendLine("            input.dispatchEvent(new KeyboardEvent('keypress', eventInit));");
        script.AppendLine("            input.dispatchEvent(new KeyboardEvent('keyup', eventInit));");
        script.AppendLine("        };");
        script.AppendLine("        setTimeout(() => {");
        script.AppendLine("            window.chrome.webview.postMessage('[LOG] 尝试: 按Enter键');");
        script.AppendLine("            pressEnter();");
        script.AppendLine("            setTimeout(() => {");
        script.AppendLine("                if (readInputValue(input).length > 0) {");
        script.AppendLine("                    window.chrome.webview.postMessage('[LOG] Enter无效，切换到点击模式');");
        script.AppendLine("                    clickButton(0);");
        script.AppendLine("                } else {");
        script.AppendLine("                    window.chrome.webview.postMessage('[LOG] ✅ Enter键成功');");
        script.AppendLine("                }");
        script.AppendLine("            }, 300);");
        script.AppendLine("        }, 100);");
        script.AppendLine("        return true;");
        script.AppendLine("    }");
        script.AppendLine("};");
        
        return script.ToString();
    }
